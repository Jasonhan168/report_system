# ─── 数据库配置（必填）─────────────────────────────────────────────────────────
# MySQL 连接字符串格式：mysql://用户名:密码@主机:端口/数据库名
DATABASE_URL=mysql://root:your_password@localhost:3306/report_db

# ─── 认证模式（必填）──────────────────────────────────────────────────────────
# hybrid = 统一登录表单，严格按 users.authType 使用 AD 或本地密码
# local / ldap 兼容统一表单；oauth 保留已有 OAuth 入口
# AD 自动注册默认关闭，在系统配置中开启后，新账号仍无任何业务权限
# AD 地址兼容 ldap://域名:389；可选 ldaps://域名:636，不强制切换现有协议
# LDAP 不加密认证传输；启用 LDAPS 时需使用受信任且匹配域名的证书
# 上线前执行版本化迁移并核对账号回填；普通存量用户默认待授权
AUTH_MODE=hybrid

# ─── JWT 密钥（必填，生产环境请使用强随机字符串）──────────────────────────────
# 生成方法（在项目目录执行）：
#   node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"
JWT_SECRET=change-this-to-a-strong-random-secret-in-production

# ─── 服务端口（可选，默认 3000）────────────────────────────────────────────────
PORT=3000
